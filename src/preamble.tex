% Markdown # / ## / ### become chapter / section / subsection.
\titleformat{\chapter}[display]
  {\normalfont\bfseries\color{kdp-charcoal}}
  {\fontsize{36}{40}\selectfont\color{kdp-accent}\thechapter}
  {10pt}
  {\fontsize{26}{31}\selectfont\raggedright}
  [\vspace{8pt}{\color{kdp-accent}\titlerule[1.2pt]}]
\titlespacing*{\chapter}{0pt}{0pt}{22pt}
\titleformat{\section}
  {\normalfont\Large\bfseries\color{kdp-charcoal}}
  {\textcolor{kdp-accent}{\thesection}}{0.6em}{}
\titlespacing*{\section}{0pt}{16pt}{7pt}
\titleformat{\subsection}
  {\normalfont\large\bfseries\color{kdp-charcoal}}
  {\textcolor{kdp-accent}{\thesubsection}}{0.6em}{}
\titlespacing*{\subsection}{0pt}{12pt}{5pt}
% Every section starts on a new page, except the first one in a chapter
\newcommand{\sectionbreak}{\ifnum\value{section}>0 \clearpage\fi}

% Chapter title on the left, author and book label on the right.
\renewcommand{\chaptermark}[1]{\markboth{\thechapter\enspace #1}{}}

  \hypersetup{pageanchor=true}}
