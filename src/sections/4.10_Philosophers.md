## Dining philosophers

### Zadatak

Proces je pokrenuo 5 niti koje simuliraju ponašanje filozofa za stolom. Filozof neko vreme misli, uzima levu i desnu viljušku na stolu, jede ručak na tanjiru, vraća viljuške, i nastavlja da misli. Filozof ne može da uzme levu viljušku ukoliko ju je trenutno filozof levo koristi, i slično ne može da uzme desnu viljušku ukoliko ju je trenutno filozof desno koristi. Ukoliko svih pet filozofa u isto vreme pokušaju da uzmu levu viljušku, ni jedan filozof neće moći da uzme desnu viljušku, i tada nastupa deadlock. Implementirati ponašanje filozofa pomoću monitora tako da se ovaj deadlock ne može dogoditi.

U svim rešenjima filozof `id` koristi viljuške `id` i `(id + 1) % N`.

### 1. način

Rešenje gde parnim filozofima obrćemo redosled uzimanja viljušaka.

**Rešenje sa `Signal-and-Continue` disciplinom**

```cpp
#include <thread.h>
#include <monitor.h>
#define N 5
#define RIGHT(id) (((id) + 1) % N)

class Table : SC_Monitor {
    bool forkFree[N] = { true, true, true, true, true };
    cond waitForFork[N];
public:
    Table() : waitForFork {this, this, this, this, this} {
        
    }
    void acquire_fork(int which) {
        monitor {
            while (! forkFree[which]) 
                waitForFork[which].wait();
            forkFree[which] = false;
        }
    }

    void release_fork(int which) {
        monitor {
            forkFree[which] = true;
            waitForFork[which].signal();
        }
    }
} table;

void phil(int id) {
    int left = id;
    int right = RIGHT(id);
    while (true) {
        // Think
        think();
        // Acquire forks
        if (id % 2 == 0) {
            table.acquire_fork(left);
            table.acquire_fork(right);
        } else {
            table.acquire_fork(right);
            table.acquire_fork(left);
        }
        // Eat
        eat();
        // Release forks
        table.release_fork(left);
        table.release_fork(right);
    }
}

int main() {
    Thread tp[N];
    for (int i = 0; i < N; i++) 
        tp[i] = createThread(phil, i);
    for (int i = 0; i < N; i++) 
        join(tp[i]);
    return 0;
}
```

### 2. način

Rešenje gde imamo „ticket“ inicijalizovan na $N-1$, gde je $N$ broj filozofa.

**Rešenje sa `Signal-and-Continue` disciplinom**

```cpp
#include <thread.h>
#include <monitor.h>
#define N 5
#define RIGHT(id) (((id) + 1) % N)

class Table : SC_Monitor {
    bool forkFree[N] = { true, true, true, true, true };
    int  ticket = N - 1;
    cond waitForFork[N];
    cond waitForTicket;
public:
    Table() 
        : waitForFork {this, this, this, this, this}
        , waitForTicket {this} {
        
    }
    void acquire_fork(int which) {
        monitor {
            while (! forkFree[which]) 
                waitForFork[which].wait();
            forkFree[which] = false;
        }
    }

    void release_fork(int which) {
        monitor {
            forkFree[which] = true;
            waitForFork[which].signal();
        }
    }
    void acquire_ticket() {
        monitor {
            while (ticket == 0)
                waitForTicket.wait();
            ticket -= 1;
        }
    }
    void release_ticket() {
        monitor {
            ticket += 1;
            waitForTicket.signal();
        }
    }
} table;

void phil(int id) {
    int left = id;
    int right = RIGHT(id);
    while (true) {
        // Think
        think();
        // Acquire forks
        table.acquire_ticket();
        table.acquire_fork(left);
        table.acquire_fork(right);
        // Eat
        eat();
        // Release forks
        table.release_fork(left);
        table.release_fork(right);
        table.release_ticket();
    }
}

int main() {
    Thread tp[N];
    for (int i = 0; i < N; i++) 
        tp[i] = createThread(phil, i);
    for (int i = 0; i < N; i++) 
        join(tp[i]);
    return 0;
}
```

### 3. način

Rešenje gde filozof uzima obe viljuške u isto vreme. Ukoliko obe viljuške nisu slobodne, filozof najavljuje da je gladan (preko `hungry[]`) i čeka. Filozof koji vraća viljuške poziva `test()` za oba suseda: ukoliko je sused gladan i obe njegove viljuške su slobodne, `test()` uzima viljuške umesto njega i budi ga. Probuđeni filozof zato ne mora ponovo da proverava da li su viljuške slobodne, pa je rešenje isto za sve tri discipline.

**Rešenje sa `Signal-and-Continue` disciplinom**

```cpp
#include <thread.h>
#include <monitor.h>
#define N 5
#define LEFT(id) (((id) + N - 1) % N)
#define RIGHT(id) (((id) + 1) % N)

class Table : SC_Monitor {
    bool forkFree[N] = { true, true, true, true, true };
    bool hungry[N] = { false, false, false, false, false };
    cond waitForForks[N];
    void test(int id) {
        if (hungry[id] && forkFree[id] && forkFree[RIGHT(id)]) {
            hungry[id] = false;
            forkFree[id] = false;
            forkFree[RIGHT(id)] = false;
            waitForForks[id].signal();
        }
    }
public:
    Table() : waitForForks {this, this, this, this, this} {
        
    }
    void acquire_forks(int id) {
        monitor {
            if (forkFree[id] && forkFree[RIGHT(id)]) {
                forkFree[id] = false;
                forkFree[RIGHT(id)] = false;
            } else {
                hungry[id] = true;
                waitForForks[id].wait();
            }
        }
    }

    void release_forks(int id) {
        monitor {
            forkFree[id] = true;
            forkFree[RIGHT(id)] = true;
            test(LEFT(id));
            test(RIGHT(id));
        }
    }
} table;

void phil(int id) {
    while (true) {
        // Think
        think();
        // Acquire forks
        table.acquire_forks(id);
        // Eat
        eat();
        // Release forks
        table.release_forks(id);
    }
}

int main() {
    Thread tp[N];
    for (int i = 0; i < N; i++) 
        tp[i] = createThread(phil, i);
    for (int i = 0; i < N; i++) 
        join(tp[i]);
    return 0;
}
```
