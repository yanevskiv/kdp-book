## Shortest job next

### Zadatak

Proces je pokrenuo $N$ niti koje treba pristupe kritičnoj sekciji po prioritetnom redosledu. Naime, ukoliko postoje više niti koje istovremeno žele da uđu u kritičnu sekciju, prva ulazi ona koja ima najmanje posla. Implementirati ovakvo ponašanje niti pomoću monitora.

**Rešenje sa `Signal-and-Wait` disciplinom**

```cpp
#include <thread.h>
#include <stdlib.h>
#include <monitor.h>
#define N 5

class SJN : SW_Monitor {
    bool locked = false;
    cond delay;
public:
    SJN() : delay(this) {
        // empty
    }
    void enter_section(int prio) {
        monitor {
            if (locked) 
                delay.wait(prio);
            locked = true;
        }
    }
    void exit_section() {
        monitor {
            locked = false;
            delay.signal();
        }
    }
} sjn;


void a(int id) {
    int prio = rand() % 1000; // nasumicni prioritet
    while (true) {
        sjn.enter_section(prio);
        work();
        sjn.exit_section();
    }
}

int main() {
    Thread ta[N];
    for (int i = 0; i < N; i++) 
        ta[i] = createThread(a, i);
    for (int i = 0; i < N; i++) 
        join(ta[i]);
    return 0;
}
```

**Rešenje sa `Signal-and-Continue` disciplinom**

```cpp
#include <thread.h>
#include <stdlib.h>
#include <monitor.h>
#define N 5

class SJN : SC_Monitor {
    bool locked = false;
    cond delay;
public:
    SJN() : delay(this) {
        // empty
    }
    void enter_section(int prio) {
        monitor {
            if (locked || delay.queue()) 
                delay.wait(prio);
            locked = true;
        }
    }
    void exit_section() {
        monitor {
            locked = false;
            delay.signal();
        }
    }
} sjn;

void a(int id) {
    int prio = rand() % 1000; // nasumicni prioritet
    while (true) {
        sjn.enter_section(prio);
        work();
        sjn.exit_section();
    }
}

int main() {
    Thread ta[N];
    for (int i = 0; i < N; i++) 
        ta[i] = createThread(a, i);
    for (int i = 0; i < N; i++) 
        join(ta[i]);
    return 0;
}
```
