## Readers and writers

### Zadatak

Proces ima jedan otvoren deljeni fajl i pokrenuo je $NR$ niti čitalaca koje izvršavaju funkciju `reader(int id)` i $NW$ niti pisaca koje izvršavaju funkciju `writer(int id)`. Niti čitaoci mogu istovremeno čitati iz fajla, dok niti pisci moraju imati ekskluzivno pravo pristupa fajlu. Ne sme se dogoditi da čitalac i pisac pristupaju fajlu u isto vreme, i ne sme se dogoditi da dva pisca upisuju u fajl u isto vreme.

U sva tri rešenja niti ulaze po FIFO redosledu: nit koja stiže čeka ukoliko neko već čeka u redu `delay`, a `delayType` pamti da li je nit u redu čitalac ili pisac. Kod `Signal-and-Continue` discipline, nit koja budi druge niti sama povećava njihove brojače, jer signalizirane niti čekaju u redu za ulazak u monitor. Kod `Signal-and-Wait` i `Signal-and-Urgent-Wait` discipline signalizirana nit odmah ulazi u monitor, pa sama povećava svoj brojač, a probuđeni čitalac budi sledećeg čitaoca iz reda.

**Rešenje sa `Signal-and-Continue` disciplinom**

```cpp
#include <thread.h>
#include <monitor.h>
#include <queue>
#define NR 10
#define NW 5
enum {
    READER = 0,
    WRITER = 1
};
class RW : SC_Monitor {
    int readerCount = 0;
    int writerCount = 0;
    std::queue<int> delayType;
    cond delay;
public:
    RW() : delay(this) {

    }
    void lock_reading() {
        monitor {
            if (writerCount > 0 || delay.queue()) {
                delayType.push(READER);
                delay.wait();
            } else {
                readerCount += 1;
            }
        }
    }
    void lock_writing() {
        monitor {
            if (writerCount > 0 || readerCount > 0 || delay.queue()) {
                delayType.push(WRITER);
                delay.wait();
            } else {
                writerCount += 1;
            }
        }
    }
    void unlock() {
        monitor {
            if (writerCount > 0) {
                writerCount -= 1;
            } else if (readerCount > 0) {
                readerCount -= 1;
            }
            if (delay.queue()) {
                if (delayType.front() == READER && writerCount == 0)  {
                    while (! delayType.empty() && delayType.front() == READER)  {
                        readerCount += 1;
                        delayType.pop();
                        delay.signal();
                    }
                } else if (delayType.front() == WRITER && readerCount == 0 && writerCount == 0) {
                    writerCount += 1;
                    delayType.pop();
                    delay.signal();
                }
            }
        }
    }
} rw;

FILE* fShared = ...;
void reader(int id) {
    while (true) {
        rw.lock_reading();
        read(fShared, ...);
        rw.unlock();
    }
}

void writer(int id) {
    while (true) {
        rw.lock_writing();
        write(fShared, ...);
        rw.unlock();
    }
}

int main() {
    Thread tw[NW], tr[NR];
    for (int i = 0; i < NR; i++) 
        tr[i] = createThread(reader, i);
    for (int i = 0; i < NW; i++) 
        tw[i] = createThread(writer, i);
    for (int i = 0; i < NR; i++) 
        join(tr[i]);
    for (int i = 0; i < NW; i++) 
        join(tw[i]);
    return 0;
}
```

**Rešenje sa `Signal-and-Wait` disciplinom**

```cpp
#include <thread.h>
#include <monitor.h>
#include <queue>
#define NR 10
#define NW 5
enum {
    READER = 0,
    WRITER = 1
};
class RW : SW_Monitor {
    int readerCount = 0;
    int writerCount = 0;
    std::queue<int> delayType;
    cond delay;
public:
    RW() : delay(this) {

    }
    void lock_reading() {
        monitor {
            if (writerCount > 0 || delay.queue()) {
                delayType.push(READER);
                delay.wait();
            }
            readerCount += 1;
            if (delay.queue() && delayType.front() == READER)  {
                delayType.pop();
                delay.signal();
            }
        }
    }
    void lock_writing() {
        monitor {
            if (writerCount > 0 || readerCount > 0 || delay.queue()) {
                delayType.push(WRITER);
                delay.wait();
            }
            writerCount += 1;
        }
    }
    void unlock() {
        monitor {
            if (writerCount > 0) {
                writerCount -= 1;
            } else if (readerCount > 0) {
                readerCount -= 1;
            }
            if (delay.queue()) {
                bool shouldUndelay =
                    (delayType.front() == READER && writerCount == 0)
                 || (delayType.front() == WRITER && writerCount == 0 && readerCount == 0);
                if (shouldUndelay) {
                    delayType.pop();
                    delay.signal();
                }
            }
        }
    }
} rw;

void reader(int id) {
    while (true) {
        rw.lock_reading();
        read(fShared, ...);
        rw.unlock();
    }
}

void writer(int id) {
    while (true) {
        rw.lock_writing();
        write(fShared, ...);
        rw.unlock();
    }
}

    for (int i = 0; i < NW; i++) 
        join(tw[i]);
    return 0;
}
```
