\section{\latin Mutual exclusion}
\textbf{\large Zadatak} Proces je pokrenuo dve niti, gde obe rade neki posao \texttt{work()} koji se mora obavljati sa ekskluzivnim pravom pristupa (npr. upis u fajl). Obezbediti ovaj vid sinhronizacije pomo\'{c}u monitora. 
\\\\
\textbf{Re\v{s}enje sa \texttt{Signal-and-Wait} disciplinom}
\begin{lstlisting}
#include <thread.h>
#include <monitor.h>
class CriticalSection : SW_Monitor {
    bool locked = false;
    cond delay;
public:
    CriticalSection() : delay(this) {

    }
    void enter_section() {
        monitor {
            if (locked) 
                delay.wait();
            locked = true;
        }
    }

    void exit_section() {
        monitor {
            locked = false;
            delay.signal();
        }
    }

} cs;

char buffer[BUFSIZ];
void a() {
    while (true) {
        cs.enter_section();
        memset(buffer, 'a', BUFSIZ);
        cs.exit_section();
        usleep(100);
    }
}

void b() {
    while (true) {
        cs.enter_section();
        memset(buffer, 'b', BUFSIZ);
        cs.exit_section();
        usleep(10);
    }
}

int main() {
    Thread ta = createThread(a);
    Thread tb = createThread(b);
    join(ta);
    join(tb);
    return 0;
}

\end{lstlisting}

#include <thread.h>
#include <monitor.h>

class CriticalSection : SC_Monitor {
    bool locked = false;
    cond delay;
public:
    CriticalSection() : delay(this) {

    }
    void enter_section() {
        monitor {
            while (locked) 
                delay.wait();
            locked = true;
        }
    }

    void exit_section() {
        monitor {
            locked = false;
            delay.signal();
        }
    }

} cs;

char buffer[BUFSIZ];
void a() {
    while (true) {
        cs.enter_section();
        memset(buffer, 'a', BUFSIZ);
        cs.exit_section();
    }
}

void b() {
    while (true) {
        cs.enter_section();
        memset(buffer, 'b', BUFSIZ);
        cs.exit_section();
    }
}

int main() {
    Thread ta = createThread(a);
    Thread tb = createThread(b);

    join(tb);
    return 0;
}

\end{lstlisting}
