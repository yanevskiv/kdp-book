## Sleeping barber

### Zadatak

Proces je pokrenuo jednu nit koja simulira ponašanje uspavanog berberina i $N$ niti koje simuliraju ponašanje mušterija koje dolaze kod berberina da se šišaju. Berberin spava dok mu ne dođe mušterija u berbernicu. Mušterija kada dođe u berbernicu treba da probudi berberina i sedne u red čekanja dok je berberin ne pozove na šišanje. Kada berberin završi sa šišanjem mušterije, mušterja plaća berberinu i odlazi. Ukoliko u berbernici nema mesta za sedenje, mušterija odlazi i vraća se nakon nekog vremena. Pomoću monitora, implementirati funkcije `void barber()` i `void customer(int id)` koje oponašaju berberina i mušterije respektivno.

**Rešenje sa `Signal-and-Wait` disciplinom** (isto rešenje radi i sa `Signal-and-Continue` disciplinom)

```cpp
#include <thread.h>
#include <monitor.h>
#include <queue>
#define MAX_SEATS 5
#define SHOP_PRICE 400
#define N 10

class BarberShop : SW_Monitor {
    std::queue<int> seats;
    int  money = 0;
    bool paid = false;
    cond waitForCustomer;
    cond waitForBarber;
    cond waitForPayment;
public:
    BarberShop() : 
        waitForCustomer(this), 
        waitForBarber(this),
        waitForPayment(this) 
    {
        // empty
    }
    int barber_wait() {
        monitor {
            if (seats.empty())
                waitForCustomer.wait();
            int custId = seats.front();
            seats.pop();
            waitForBarber.signal();
            return custId;
        }
    }

    int barber_done(int id) {
        monitor {
            if (! paid) 
                waitForPayment.wait();
            int payment = money;
            paid = false;
            money = 0;
            return payment;
        }
    }

    bool enter_shop(int id) {
        monitor {
            if (seats.size() < MAX_SEATS) {
                seats.push(id);
                waitForCustomer.signal();
                waitForBarber.wait();
                return true;
            }
            return false;
        }
    }

    void exit_shop(int payment) {
        monitor {
            paid = true;
            money += payment;
            waitForPayment.signal();
        }
    }
} shop;

void barber() {
    int totalMoney = 0;
    while (true) {
        int custId = shop.barber_wait();
        /* cut hair */
        totalMoney += shop.barber_done(custId);
    }
}

void customer(int id) {
    while (true) {
        if (shop.enter_shop(id)) {
            /* cut hair */
            shop.exit_shop(SHOP_PRICE);
        }
        usleep(50);
    }
}

int main() {
    Thread tb = createThread(barber);
    Thread tc[N];
    for (int i = 0; i < N; i++) 
        tc[i] = createThread(customer, i);
    join(tb);
    join(tb);
    return 0;
}
```
